function ToyView({ toy: initial, back }) {
  const [toy, setToy] = useState(initial);
  const [tab, setTab] = useState(initial.date ? "guests" : "toy");
  const [d, setD] = useState({ guests: [], rsvps: [], artists: [], expenses: [], gifts: [] });
  const [err, setErr] = useState("");
  const [loading, setLoading] = useState(true);

  const load = async () => {
    const q = (t) => sb.from(t).select("*").eq("toy_id", toy.id).order(t === "artists" || t === "expenses" ? "id" : "created_at");
    const res = await Promise.all(["guests","rsvps","artists","expenses","gifts"].map(q));
    const bad = res.find((r) => r.error); if (bad) setErr(errText(bad.error));
    setD({ guests: res[0].data || [], rsvps: res[1].data || [], artists: res[2].data || [], expenses: res[3].data || [], gifts: res[4].data || [] });
    setLoading(false);
  };
  useEffect(() => { load(); const i = setInterval(load, 30000); return () => clearInterval(i); }, []);

  // действия с базой: после каждого — перезагрузка
  const act = async (promise) => { const { error } = await promise; if (error) setErr(errText(error)); else setErr(""); await load(); };
  const db = {
    add: (t, rows) => act(sb.from(t).insert((Array.isArray(rows) ? rows : [rows]).map((r) => ({ ...r, toy_id: toy.id })))),
    upd: (t, id, patch) => act(sb.from(t).update(patch).eq("id", id)),
    del: (t, id) => act(sb.from(t).delete().eq("id", id)),
  };

  // сопоставление ответов гостей со списком
  const guests = d.guests.map((g) => {
    const gt = tokens(g.name);
    const m = d.rsvps.find((r) => { const rt = tokens(r.name); return gt.length && gt.every((x) => rt.includes(x)); })
      || d.rsvps.find((r) => gt[0] && tokens(r.name).includes(gt[0]));
    return m ? { ...g, status: m.come ? "yes" : "no", count: m.come ? m.count : g.count, self: m } : g;
  });
  const used = new Set(guests.filter((g) => g.self).map((g) => g.self.id));
  const strangers = d.rsvps.filter((r) => !used.has(r.id));
  const yes = guests.filter((g) => g.status === "yes").reduce((s, g) => s + Number(g.count || 0), 0) + strangers.filter((r) => r.come).reduce((s, r) => s + r.count, 0);
  const invited = d.guests.reduce((s, g) => s + Number(g.count || 1), 0);
  const plan = d.expenses.reduce((s, x) => s + Number(x.plan), 0) + d.artists.reduce((s, a) => s + Number(a.fee) + Number(a.travel), 0);
  const paid = d.expenses.reduce((s, x) => s + Number(x.paid), 0) + d.artists.reduce((s, a) => s + Number(a.paid), 0);
  const days = toy.date ? Math.ceil((new Date(toy.date) - new Date()) / 86400000) : null;
  const stat = (big, small) => html`<div><div style="font-family:Lora,serif;font-size:20px;font-weight:700">${big}</div><div style="opacity:.75">${small}</div></div>`;
  const TABS = [["toy","Той"],["guests","Гости"],["program","Программа"],["budget","Бюджет"],["gifts","Кәде дәптері"]];

  return html`<div>

        <h1 style="font-family:Lora,serif;font-size:26px;margin:4px 0 12px">${toy.names}</h1>
        <div style="display:flex;gap:22px;flex-wrap:wrap;font-size:14px">
          ${stat(days === null ? "—" : days >= 0 ? days : "прошёл", "дней до тоя")}
          ${stat(`${yes} / ${invited}`, "придут / приглашено")}
          ${stat(fmt(plan - paid), "осталось оплатить")}
        </div>
      </div>
    </header>
    <nav class="tabs">${TABS.map(([k, l]) => html`<button class=${tab === k ? "on" : ""} onClick=${() => setTab(k)}>${l}</button>`)}</nav>
    <div class="wrap">
      ${err && html`<p class="err">${err}</p>`}
      ${loading ? html`<p>Загружаю…</p>` : html`
        ${tab === "toy" && html`<${ToySettings} toy=${toy} setToy=${setToy} />`}

function Guests({ toy, guests, strangers, db, reload }) {
  const [g, setG] = useState({ name: "", phone: "", count: 2 });
  const [bulk, setBulk] = useState(""); const [showBulk, setShowBulk] = useState(false);
  const [filter, setFilter] = useState("all");
  const add = () => { if (!g.name.trim()) return; db.add("guests", { name: g.name.trim(), phone: g.phone, count: Number(g.count) || 1 }); setG({ name: "", phone: "", count: 2 }); };
  const importBulk = () => {
    const have = new Set(guests.map((x) => norm(x.name).replace(/[\s–-]/g, "")));
    const rows = cleanLines(bulk).map((n) => ({ name: n.replace(/\s*-\s*/g, " – "), count: /-/.test(n) ? 2 : 1 })).filter((r) => !have.has(norm(r.name).replace(/[\s–-]/g, "")));
    if (rows.length) db.add("guests", rows);
    setBulk(""); setShowBulk(false);
  };
  const invite = (x) => {
    const text = `Құрметті ${x.name}!\nСіздерді ${genitive(toy.names)} ${toy.kind.toLowerCase()}ына шақырамыз.\nШақыру және келетініңізді растау осы сілтемеде:\n${inviteUrl(toy.slug)}`;
    const phone = (x.phone || "").replace(/\D/g, "").replace(/^8/, "7");
    window.open(`https://wa.me/${phone}?text=${encodeURIComponent(text)}`, "_blank");
  };
  const cnt = (s) => guests.filter((x) => x.status === s).length;
  const shown = guests.filter((x) => filter === "all" || x.status === filter);
  const badge = (x) => x.status === "yes" ? html`<span class="badge" style="background:#2E7D4F1A;color:var(--green)">придут · ${x.count} чел.</span>`
    : x.status === "no" ? html`<span class="badge" style="background:#B5483B1A;color:var(--red)">не придут</span>`
    : html`<span class="badge" style="background:#6B8A8E1A;color:#6B8A8E">ждём · ${x.count} чел.</span>`;
  return html`
    <h2>Гости</h2>
    <p class="hint">Отправьте гостю приглашение в WhatsApp. Когда он ответит, статус обновится здесь сам.</p>
    <input value=${g.name} onInput=${(e) => setG({ ...g, name: e.target.value })} placeholder="Имя гостя или пары, например Берик-Айэстра"/>
    <div style="display:flex;gap:8px;margin-top:8px">
      <input value=${g.phone} onInput=${(e) => setG({ ...g, phone: e.target.value })} placeholder="+7 7__ ___ __ __" inputmode="tel"/>
      <input type="number" min="1" value=${g.count} onInput=${(e) => setG({ ...g, count: e.target.value })} style="width:80px" aria-label="Сколько человек"/>
    </div>
    <button class="btn" onClick=${add}>Добавить гостя</button>
    <button class="btn ghost" onClick=${() => setShowBulk(!showBulk)}>Вставить список из WhatsApp</button>
    ${showBulk && html`<div style="margin-top:8px">
      <textarea rows="8" value=${bulk} onInput=${(e) => setBulk(e.target.value)} placeholder=${"1. Берик-Айэстра\n2. Арайлым"}></textarea>
      <p class="hint">Пара через дефис считается как 2 человека. Повторы пропускаются.</p>
      <button class="btn" onClick=${importBulk}>Добавить всех</button></div>`}
    <button class="btn ghost" onClick=${reload}>Обновить ответы гостей</button>
    <div class="chips">${[["all",`Все ${guests.length}`],["yes",`Придут ${cnt("yes")}`],["no",`Не придут ${cnt("no")}`],["wait",`Не ответили ${cnt("wait")}`]].map(([k, l]) =>
      html`<button class=${filter === k ? "on" : ""} onClick=${() => setFilter(k)}>${l}</button>`)}</div>
    ${shown.map((x) => html`<div class="row" key=${x.id}><div>
      <div style="display:flex;justify-content:space-between;gap:8px;align-items:center"><b>${x.name}</b>${badge(x)}</div>
      ${x.self ? html`<div class="hint" style="margin:4px 0 0">Ответили сами как «${x.self.name}»${x.self.wish ? ` · «${x.self.wish}»` : ""}</div>` : html`
        <div style="display:flex;gap:6px;margin-top:8px">
          <input value=${x.phone} placeholder="Телефон" inputmode="tel" style="padding:6px" onChange=${(e) => db.upd("guests", x.id, { phone: e.target.value })}/>
          <button onClick=${() => invite(x)} disabled=${!x.phone} style=${`border:none;border-radius:8px;padding:6px 12px;color:#fff;white-space:nowrap;background:${x.phone ? "#25D366" : "#B9D9C4"}`}>Отправить</button>
        </div>`}
    </div><button class="x" aria-label="Удалить" onClick=${() => confirm("Удалить " + x.name + "?") && db.del("guests", x.id)}>✕</button></div>`)}
    ${strangers.length > 0 && html`
      <h2>Ответили, но нет в списке</h2>
      <p class="hint">Проверьте, может быть гость написал имя по-другому.</p>
      ${strangers.map((r) => html`<div class="row" key=${r.id}><div><b>${r.name}</b> · ${r.come ? `придут, ${r.count} чел.` : "не придут"}${r.wish ? html`<div class="hint" style="margin:0">«${r.wish}»</div>` : ""}</div>
        <button class="btn ghost" style="width:auto;margin:0;padding:6px 10px" onClick=${() => db.add("guests", { name: r.name, count: r.count || 1 })}>В список</button></div>`)}`}`;
}

function Program({ list, db }) {
  const [bulk, setBulk] = useState(""); const [role, setRole] = useState("Артист");
  const importBulk = () => {
    const rows = cleanLines(bulk).map((l) => { const m = l.match(/^(.*?)\s*\(([^)]*)\)\s*$/); return { role, name: (m ? m[1] : l).trim(), city: m ? m[2].trim() : "" }; });
    if (rows.length) db.add("artists", rows); setBulk("");
  };
  const num = (x, k, ph) => html`<input type="number" value=${x[k] || ""} placeholder=${ph} style="padding:6px;font-size:14px" onChange=${(e) => db.upd("artists", x.id, { [k]: Number(e.target.value) || 0 })}/>`;
  const noPrice = list.filter((x) => !Number(x.fee)).length;
  return html`
    <h2>Программа</h2>
    <p class="hint">Ведущие и артисты. Гонорар, дорога с проживанием и оплата сразу идут в бюджет.</p>
    ${noPrice > 0 && html`<p class="err">Без суммы: ${noPrice}</p>`}
    ${["Ведущий","Артист"].map((r) => html`<div>
      <h3 style="font-size:16px;margin:18px 0 0">${r === "Ведущий" ? "Ведущие" : "Артисты и коллективы"}</h3>
      ${list.filter((x) => x.role === r).map((x) => html`<div class="row" key=${x.id}><div>
        <b>${x.name}</b> <span style="color:#6B8A8E">· ${x.city}</span>
        <div style="display:grid;grid-template-columns:1fr 1fr 1fr;gap:6px;margin-top:8px">${num(x, "fee", "Гонорар")}${num(x, "travel", "Дорога")}${num(x, "paid", "Оплачено")}</div>
      </div><button class="x" aria-label="Удалить" onClick=${() => confirm("Удалить " + x.name + "?") && db.del("artists", x.id)}>✕</button></div>`)}
    </div>`)}
    <h2>Добавить списком</h2>
    <p class="hint">Каждый с новой строки, город в скобках, как в WhatsApp: «Номат тобы (Тараз)».</p>
    <select value=${role} onChange=${(e) => setRole(e.target.value)}><option>Артист</option><option>Ведущий</option></select>
    <textarea rows="6" value=${bulk} onInput=${(e) => setBulk(e.target.value)} style="margin-top:8px" placeholder=${"1. Мая Веронская (Алматы)\n2. Марат Алибаев (Алматы)"}></textarea>
    <button class="btn" onClick=${importBulk}>Добавить</button>`;
}

function Host() {
  const [session, setSession] = useState(undefined);
  const [toy, setToy] = useState(null);
  useEffect(() => {
    sb.auth.getSession().then(({ data }) => setSession(data.session));
    const { data } = sb.auth.onAuthStateChange((_e, s) => setSession(s));
    return () => data.subscription.unsubscribe();
  }, []);
  if (session === undefined) return html`<p style="padding:40px">Загружаю…</p>`;
  if (!session) return html`<${Login} />`;
  return toy ? html`<${ToyView} toy=${toy} back=${() => setToy(null)} />` : html`<${ToyList} open=${setToy} />`;
}
